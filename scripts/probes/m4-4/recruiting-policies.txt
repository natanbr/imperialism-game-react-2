const PROBE_R = Number(process.env.PROBE_R ?? 0);          // 1: recruit from stock only; 2: also make and keep the goods
const PROBE_ALWAYS = !!process.env.PROBE_ALWAYS;            // recruit even when the run leaves labour idle
function PROBE_planRecruiting(view, nationId, propose) {
  if (!PROBE_R) return;
  const state = view();
  if (turnsAfter(state) < 8) return;
  if (!PROBE_ALWAYS && planRun(state, nationId).labour >= 2) return;      // FR-130's "labour-bound" test
  const nation = nationOf(state, nationId); const stock = nation.warehouse; const mouths = mouthsOf(nation);
  if ((stock.cannedFood ?? 0) < mouths + 1) return;                     // a meal of cans stays
  const gathered = gatheredOf(state, nationId); const needs = mealNeeds(mouths + 1);
  if (gathered.grain + stock.grain < needs.grain || gathered.fruit + stock.fruit < needs.fruit
      || gathered.livestock + stock.livestock < needs.meat) return;     // the new mouth can be fed
  propose({ kind: "recruitWorkers", nationId, delta: 1 });              // validation needs clothing + furniture + can
}
